// =========================================================
//  Binary Space Partition Tree
// =========================================================

typedef struct BSP_Node
{
    Rectangle rect;
    struct BSP_Node *left;
    struct BSP_Node *right;
} BSP_Node;

BSP_Node *create_node(Rectangle rect, BSP_Node *left, BSP_Node *right)
{
    BSP_Node *node = malloc(sizeof(BSP_Node));
    node->rect = rect;
    node->left = left;
    node->right = right;
    return node;
}

Vector2 get_rand_vec2(Vector2 min, Vector2 max)
{
    Vector2 result;
    result.x = (float)rand() / (float)RAND_MAX * (max.x - min.x) + min.x;
    result.y = (float)rand() / (float)RAND_MAX * (max.y - min.y) + min.y;
    return result;
}

int get_tile(int *map, int x, int y, Vector2 subdivisions)
{
    // Safety: check bounds before array access
    if (x < 0 || y < 0 || x >= subdivisions.x || y >= subdivisions.y)
        return -1;
    return map[y * (int)subdivisions.x + x];
}

                x * tile_size.x,
                y * tile_size.y,
                tile_size.x - 1,
                tile_size.y - 1,

        }
    }
}

float ray_march(Vector2 origin, Vector2 dir, int *map, Vector2 subdivisions, Vector2 tile_size)
{
    float max_dist = 1000.0f;
    float step = 2.0f; // Precision vs Performance trade-off

    for (float d = 0; d < max_dist; d += step)
    {
        Vector2 p = Vector2Add(origin, Vector2Scale(dir, d));
        if (get_tile(map, p.x / tile_size.x, p.y / tile_size.y, subdivisions) == WALL)
        {
            return d;
        }
    }
    return max_dist;
}

float ray_dda(Vector2 pos, Vector2 dir, int *map, Vector2 subdivisions, Vector2 world_tile_size)
{
    float hit_dist;
    // truncate to tile
    int current_col = (int)(pos.x / world_tile_size.x);
    int current_row = (int)(pos.y / world_tile_size.y);

    // ray length from pos to tile normlized
    float delta_x = (dir.x == 0) ? INFINITY : fabsf(1.0f / dir.x);
    float delta_y = (dir.y == 0) ? INFINITY : fabsf(1.0f / dir.y);

    int step_x = dir.x < 0 ? -1 : 1;
    int step_y = dir.y < 0 ? -1 : 1;

    float side_dist_x;
    float side_dist_y;

    if (dir.x < 0)
    {
        // side_dist_x = (pos_col - map_col) * normal_dir_x
        // Notice the float vs int here, always a remainder
        side_dist_x = ((float)(pos.x / world_tile_size.x) - (int)(current_col)) * delta_x;
    }
    else
    {
        int next_col = current_col + 1;
        side_dist_x = (float)(next_col - (float)(pos.x / world_tile_size.x)) * delta_x;
    }

    if (dir.y < 0)
    {
        side_dist_y = (float)((float)(pos.y / world_tile_size.y) - (int)(current_row)) * delta_y;
    }
    else
    {
        int next_row = current_row + 1;
        side_dist_y = (float)(next_row - (float)(pos.y / world_tile_size.y)) * delta_y;
    }
    // basically just jump in the direction ray is going more in

    int hit = 0;
    int side;
    while (!hit)
    {
        // Example safety check in your get_tile
        if (current_col < 0 || current_col >= subdivisions.x || current_row < 0 || current_row >= subdivisions.y)
            return WALL;
        if (side_dist_x < side_dist_y)
        {
            side_dist_x += delta_x;
            current_col += step_x;
            side = 0;
        }
        else
        {
            side_dist_y += delta_y;
            current_row += step_y;
            side = 1;
        }
        if (get_tile(map, current_col, current_row, subdivisions) == WALL)
        {
            hit = 1;
        }
    }
    if (side == 0)
    {
        hit_dist = side_dist_x - delta_x;
        hit_dist *= world_tile_size.x;
    }
    else
    {
        hit_dist = side_dist_y - delta_y;
        hit_dist *= world_tile_size.y;
    }
    // Safety break for infinite voids
    if (hit_dist > 1000.0f)
        return 1000.0f;

    return hit_dist;
}
void draw_ray(Vector2 origin, Vector2 direction, float t, Color c)
{
    // r = origin + direction * t
    if (Vector2Length(direction) > 1.0f)
    {
        direction = Vector2Normalize(direction);
    }
    float length = t;
    DrawLineEx(origin, Vector2Add(origin, Vector2Scale(direction, length)), 2.0f, c);
}
void draw_player_forwardv(Vector2 center, Vector2 forward_v, float length, Color c)
{
    // The end point is: Center + (Direction * Length)
    float lineLength = length;
    Vector2 lineEnd = Vector2Add(center, Vector2Scale(forward_v, lineLength));

    DrawLineEx(center, lineEnd, 2.0f, c);
}

    *u = Vector2Add(*u, Vector2Scale(v, dt));

int in_bounds(Player *player, int *map, Vector2 subdivisions, Vector2 world_tile_size)
{
    int x = player->position.x / world_tile_size.x;
    int y = player->position.y / world_tile_size.y;
    int tile = get_tile(map, x, y, subdivisions);
    return (tile == EMPTY);
}

void update_player_orientation(Player *player, Vector2 velocity, float scalar, float dt)
{
    if (velocity.x == 0 && velocity.y == 0) {return;}
    
    // atan2f gives the angle in radians for any vector
    player->angle_radians = atan2f(velocity.y, velocity.x);
    if (player->angle_radians < 0) player->angle_radians += 2 * M_PI;

    // Forward vector is a simple unit vector based on that angle
    player->forward_v.x = cosf(player->angle_radians);
    player->forward_v.y = sinf(player->angle_radians);
}

void move_and_slide(Player *p, Vector2 velocity, int *map, Vector2 subdivisions, Vector2 world_tile_size)
{
    // --- 1. RESOLVE X-AXIS ---
    float next_x = p->position.x + velocity.x;

    // Determine which edge to check based on direction
    // If moving right (pos), check the right edge (x + width).
    // If moving left (neg), check the left edge (x).
    float check_x = (velocity.x > 0) ? (next_x + p->size.x) : next_x;

    // Check both the top and bottom corners of that leading edge
    int tile_x_top = get_tile(map, (int)(check_x / world_tile_size.x), (int)(p->position.y / world_tile_size.y), subdivisions);
    int tile_x_bottom = get_tile(map, (int)(check_x / world_tile_size.x), (int)((p->position.y + p->size.y - 1) / world_tile_size.y), subdivisions);

    if (tile_x_top == EMPTY && tile_x_bottom == EMPTY)
    {
        p->position.x = next_x;
    }

    // --- 2. RESOLVE Y-AXIS ---
    float next_y = p->position.y + velocity.y;

    // If moving down (pos), check bottom edge (y + height).
    // If moving up (neg), check top edge (y).
    float check_y = (velocity.y > 0) ? (next_y + p->size.y) : next_y;

    // Check both the left and right corners of that leading edge
    int tile_y_left = get_tile(map, (int)(p->position.x / world_tile_size.x), (int)(check_y / world_tile_size.y), subdivisions);
    int tile_y_right = get_tile(map, (int)((p->position.x + p->size.x - 1) / world_tile_size.x), (int)(check_y / world_tile_size.y), subdivisions);

    if (tile_y_left == EMPTY && tile_y_right == EMPTY)
    {
        p->position.y = next_y;
    }
}

void controls(Player *p, int *map, Vector2 subdivisions, Vector2 world_tile_size, float move_speed)
{
    float dt = GetFrameTime();
    Vector2 vel = {0, 0};

    if (IsKeyDown(KEY_W))
        vel.y -= 1;
    if (IsKeyDown(KEY_S))
        vel.y += 1;
    if (IsKeyDown(KEY_A))
        vel.x -= 1;
    if (IsKeyDown(KEY_D))
        vel.x += 1;

    if (vel.x != 0 || vel.y != 0)
    {
        vel = Vector2Scale(Vector2Normalize(vel), move_speed * dt);
        update_player_orientation(p, vel, move_speed / 50, dt);
        move_and_slide(p, vel, map, subdivisions, world_tile_size);
    }
    if (IsKeyPressed(KEY_R))
        p->position = get_screen_center();
}

void draw_render(int *map, Vector2 subdiv, Vector2 world_tile_size, Player *player)
{
    int pos_x = player->position.x / world_tile_size.x;
    int pos_y = player->position.y / world_tile_size.y;
    float height = 1/ray_dda(player->position, player->forward_v, map, subdiv, world_tile_size);
    float width = height;
    Color walls_color = BLUE;

    DrawRectangle(pos_x * world_tile_size.x, pos_y * world_tile_size.y, width, height, walls_color);
}

void manage_scissor_camera(Rectangle *scissor_rect, Camera2D *cam, int *map, Vector2 subdivisions, Player *player, Color c, Vector2 world_tile_size, int id)
{
    BeginScissorMode(scissor_rect->x, scissor_rect->y,
                     scissor_rect->width, scissor_rect->height);
    BeginMode2D(*cam);
    switch (id)
    {
    case RENDER:
        /* code */
        draw_render(map, subdivisions, world_tile_size, player);
        break;
    case RAY:
        draw_map_topdown(map, subdivisions.x, subdivisions.y, world_tile_size);
        static int get_color = 0;
        static Color forwardv_color;
        static Color ray_color;
        if (get_color == 0)
        {
            forwardv_color = get_random_color();
            ray_color = get_random_color();
            get_color = 1;
        }
    update_forwardv(player, player->angle_radians);
    draw_player(player, c);
    Vector2 center = get_player_center(player);

    float fov = 60.0f * DEG2RAD; // 60 degrees in radians
    int num_rays = fov/DEG2RAD; // One ray per degree is usually plenty for top-down
    float step = fov / num_rays; // 1degree per ray

    for (int i = 0; i < num_rays; i++)
    {
        // Calculate angle for THIS specific ray
        // We start at (center - half fov) and sweep to (center + half fov)
        float ray_angle = (player->angle_radians - (fov / 2.0f)) + (i * step);
        
        Vector2 ray_dir = { cosf(ray_angle), sinf(ray_angle) };
        
        // Find distance for this specific ray
        //float dist = ray_march(center, ray_dir, map, subdivisions, world_tile_size);
        float dist = ray_dda(center, ray_dir, map, subdivisions, world_tile_size);
        
        // Draw the ray
        draw_ray(center, ray_dir, dist, ray_color);
    }

    // Draw the main forward vector last so it's on top
    float center_dist = ray_dda(center, player->forward_v, map, subdivisions, world_tile_size);
    draw_player_forwardv(center, player->forward_v, player->size.y, BLACK);
    
    DrawText(TextFormat("Wall Distance: %.2f", center_dist), 10, 10, 20, WHITE);
    break;
    case TREE:
        draw_map_topdown(map, subdivisions.x, subdivisions.y, world_tile_size);
        draw_player(player, c);
        break;
    default:
        break;
    }

    EndMode2D();
    EndScissorMode();
}
